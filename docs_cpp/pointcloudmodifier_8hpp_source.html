132 Modifier & cropBox(const std::vector<double> & box_params, const bool & negative = false)
133 {
134 // Check box_params size
135 if (box_params.size() != 6) {
136 std::cerr << "Error: box_params must contain exactly 6 elements." << std::endl;
137 return *this;
138 }
139
140 pcl::CropBox<PointCloud> boxFilter;
141 boxFilter.setMin(Eigen::Vector4f(box_params[0], box_params[1], box_params[2], 1.0));
142 boxFilter.setMax(Eigen::Vector4f(box_params[3], box_params[4], box_params[5], 1.0));
143 boxFilter.setInputCloud(output_cloud);
144 boxFilter.setNegative(negative);
145 boxFilter.filter(*output_cloud);
146 return *this;

154 Modifier & cropSphere(const double & sphere_params, const bool & negative = false)
155 {
156 const auto radius_sq = sphere_params * sphere_params;
157 pcl::PointCloud<pcl::PointXYZ>::Ptr tmp(new pcl::PointCloud<pcl::PointXYZ>);
158 pcl::fromPCLPointCloud2(*output_cloud, *tmp);
159
160 pcl::PointIndices::Ptr inliers(new pcl::PointIndices());
161 inliers->indices.reserve(tmp->size());
162
163 for (size_t i = 0; i < tmp->size(); ++i) {
164 if (tmp->points[i].getVector3fMap().squaredNorm() < radius_sq) {
165 inliers->indices.push_back(i);
166 }
167 }
168
169 pcl::ExtractIndices<pcl::PCLPointCloud2> extract;
170 extract.setInputCloud(output_cloud);
171 extract.setIndices(inliers);
172 extract.setNegative(negative);
173 extract.filter(*output_cloud);
174
175 return *this;
176 }

183 Modifier & cropCylinder(const double & zylinder_params, const bool & negative = false)
184 {
185 const auto radius_sq = zylinder_params * zylinder_params;
186 pcl::PointCloud<pcl::PointXYZ>::Ptr tmp(new pcl::PointCloud<pcl::PointXYZ>);
187 pcl::fromPCLPointCloud2(*output_cloud, *tmp);
188
189 pcl::PointIndices::Ptr inliers(new pcl::PointIndices());
190 inliers->indices.reserve(tmp->size());
191
192 for (size_t i = 0; i < tmp->size(); ++i) {
193 auto point = tmp->points[i].getVector3fMap();
194 auto r = point.x() * point.x() + point.y() * point.y();
195 if (r < radius_sq) {
196 inliers->indices.push_back(i);
197 }
198 }
199
200 pcl::ExtractIndices<pcl::PCLPointCloud2> extract;
201 extract.setInputCloud(output_cloud);
202 extract.setIndices(inliers);
203 extract.setNegative(negative);
204 extract.filter(*output_cloud);
205
206 return *this;
207 }

213 Modifier & voxelFilter(const std::vector<double> & voxel)
214 {
215 // Check voxel size
216 if (voxel.size() != 3) {
217 std::cerr << "Error: voxel must contain exactly 3 elements." << std::endl;
218 return *this;
219 }
220
221 pcl::PointCloud<pcl::PointXYZ>::Ptr tmp(new pcl::PointCloud<pcl::PointXYZ>);
222 pcl::fromPCLPointCloud2(*output_cloud, *tmp);
223
224 // Compute min_pt and max_pt
225 pcl::PointXYZ min_pt, max_pt;
226 pcl::getMinMax3D(*tmp, min_pt, max_pt);
227
228 // Figure out how many cells we’d have in each dimension
229 const auto num_x = static_cast<uint64_t>(std::floor((max_pt.x - min_pt.x) / voxel[0]));
230 const auto num_y = static_cast<uint64_t>(std::floor((max_pt.y - min_pt.y) / voxel[1]));
231 const auto num_z = static_cast<uint64_t>(std::floor((max_pt.z - min_pt.z) / voxel[2]));
232 const auto total = num_x * num_y * num_z;
233
234 // If the total # of voxels would exceed int32_t, apply subvoxel
235 if (total > static_cast<uint64_t>(INT32_MAX)) {
236 applySubVoxelFilter(voxel, output_cloud, min_pt, max_pt, num_x, num_y, num_z);
237 } else {
238 applyVoxelFilter(voxel, output_cloud);
239 }
240
241 return *this;
242 }

287 // Transforms
293 Modifier & transform(const Eigen::Affine3d & transformation)
294 {
295 // Find x, y, z field offsets
296 int x_offset = -1, y_offset = -1, z_offset = -1;
297 for (const auto & field : output_cloud->fields) {
298 if (field.name == "x") x_offset = field.offset;
299 else if (field.name == "y") y_offset = field.offset;
300 else if (field.name == "z") z_offset = field.offset;
301 }
302
303 Eigen::Affine3f transform_f = transformation.cast<float>();
304
305 // Transform each point in-place
306 for (size_t i = 0; i < output_cloud->data.size(); i += output_cloud->point_step) {
307 float x, y, z;
308 memcpy(&x, &output_cloud->data[i + x_offset], sizeof(float));
309 memcpy(&y, &output_cloud->data[i + y_offset], sizeof(float));
310 memcpy(&z, &output_cloud->data[i + z_offset], sizeof(float));
311
312 Eigen::Vector3f pt(x, y, z);
313 Eigen::Vector3f pt_transformed = transform_f * pt;
314
315 memcpy(&output_cloud->data[i + x_offset], &pt_transformed.x(), sizeof(float));
316 memcpy(&output_cloud->data[i + y_offset], &pt_transformed.y(), sizeof(float));
317 memcpy(&output_cloud->data[i + z_offset], &pt_transformed.z(), sizeof(float));
318 }
319
320 return *this;
321 }

325 using ProgressCallback = std::function<void (const std::string &, size_t, size_t)>;
326
333 std::map<std::pair<int, int>, Modifier> split(
334 const double & grid_size,
335 const ProgressCallback & on_progress = nullptr)
336 {
337 std::map<std::pair<int, int>, Modifier> result;
338
339 if (output_cloud->data.empty()) {
340 std::cerr << "Error: Cannot split empty point cloud" << std::endl;
341 return result;
342 }
343 if (grid_size <= 0.0) {
344 std::cerr << "Error: grid_size must be positive" << std::endl;
345 return result;
346 }
347
348 int x_offset = -1, y_offset = -1;
349 for (const auto & field : output_cloud->fields) {
350 if (field.name == "x") x_offset = static_cast<int>(field.offset);
351 else if (field.name == "y") y_offset = static_cast<int>(field.offset);
352 }
353 if (x_offset == -1 || y_offset == -1) {
354 std::cerr << "Error: Point cloud must have x and y fields" << std::endl;
355 return result;
356 }
357
358 const size_t num_points = output_cloud->width * output_cloud->height;
359 const size_t point_step = output_cloud->point_step;
360
361 // Collect point indices per grid cell
362 std::map<std::pair<int, int>, std::vector<size_t>> grid_index;
363 const size_t bin_report_step = std::max<size_t>(num_points / 100, 1);
364 for (size_t i = 0; i < num_points; ++i) {
365 const uint8_t * pt_data = output_cloud->data.data() + i * point_step;
366 float x, y;
367 memcpy(&x, pt_data + x_offset, sizeof(float));
368 memcpy(&y, pt_data + y_offset, sizeof(float));
369 int ix = static_cast<int>(std::floor(x / grid_size));
370 int iy = static_cast<int>(std::floor(y / grid_size));
371 grid_index[{ix, iy}].push_back(i);
372 if (on_progress && ((i + 1) % bin_report_step == 0 || i + 1 == num_points)) {
373 on_progress("Binning points", i + 1, num_points);
374 }
375 }
376
377 const size_t total_cells = grid_index.size();
378 size_t cells_built = 0;
379 for (const auto & cell : grid_index) {
380 const auto & indices = cell.second;
381 PointCloud::Ptr cell_cloud = std::make_shared<PointCloud>();
382 cell_cloud->header = output_cloud->header;
383 cell_cloud->fields = output_cloud->fields;
384 cell_cloud->is_bigendian = output_cloud->is_bigendian;
385 cell_cloud->point_step = point_step;
386 cell_cloud->width = indices.size();
387 cell_cloud->height = 1;
388 cell_cloud->is_dense = true;
389 cell_cloud->row_step = cell_cloud->point_step * cell_cloud->width;
390 cell_cloud->data.resize(indices.size() * point_step);
391
392 for (size_t j = 0; j < indices.size(); ++j) {
393 memcpy(
394 cell_cloud->data.data() + j * point_step,
395 output_cloud->data.data() + indices[j] * point_step,
396 point_step);
397 }
398
399 Modifier cell_modifier;
400 cell_modifier.setCloud(cell_cloud);
401 result.emplace(cell.first, std::move(cell_modifier));
402
403 ++cells_built;
404 if (on_progress) {
405 on_progress("Building cells", cells_built, total_cells);
406 }
407 }
408
409 return result;
410 }

420 static bool writeGridMetadata(
421 const std::map<std::pair<int, int>, Modifier> & cells,
422 const double & grid_size,
423 const std::string & file_path,
424 const std::string & extension = ".pcd")
425 {
426 std::ofstream file(file_path);
427 if (!file.is_open()) {
428 std::cerr << "Error: Failed to write metadata file: " << file_path << std::endl;
429 return false;
430 }
431
432 auto format_coord = [](double val) -> std::string {
433 if (val == std::floor(val)) {
434 return std::to_string(static_cast<int64_t>(val));
435 }
436 std::ostringstream oss;
437 oss << val;
438 return oss.str();
439 };
440
441 file << "x_resolution: " << grid_size << "\n";
442 file << "y_resolution: " << grid_size << "\n";
443
444 for (const auto & cell : cells) {
445 double coord_x = cell.first.first * grid_size;
446 double coord_y = cell.first.second * grid_size;
447 std::string sx = format_coord(coord_x);
448 std::string sy = format_coord(coord_y);
449 file << sx << "_" << sy << extension << ": [" << sx << ", " << sy << "]\n";
450 }
451
452 return true;
453 }

455 // Analysis
461 Modifier & timestampAnalyzer(const std::string & file_path)
462 {
463 std::vector<double> time_float{};
464 std::vector<size_t> time_int{};
465 for (size_t i = 0; i < output_cloud->data.size(); i += output_cloud->point_step) {
466 for (size_t j = 0; j < output_cloud->fields.size(); ++j) {
467 double tf{0.0};
468 size_t ti{0};
469 pcl::PCLPointField & field = output_cloud->fields[j];
470 size_t point_offset = i + field.offset;
471 if (field.name == "timestamp" || field.name == "time_stamp" || field.name == "t") {
472 if (field.datatype == pcl::PCLPointField::FLOAT32) {
473 float tmp_stamp;
474 memcpy(&tmp_stamp, &output_cloud->data[point_offset], sizeof(float));
475 tf = static_cast<double>(tmp_stamp);
476 } else if (field.datatype == pcl::PCLPointField::FLOAT64) {
477 memcpy(&tf, &output_cloud->data[point_offset], sizeof(double));
478 } else if (field.datatype == pcl::PCLPointField::UINT32) {
479 memcpy(&ti, &output_cloud->data[point_offset], sizeof(uint32_t));
480 } else if (field.datatype == pcl::PCLPointField::UINT8 && field.count == 8) {
481 ti = 0;
482 for (size_t k = 0; k < 8; ++k) {
483 ti |= static_cast<uint64_t>(output_cloud->data[point_offset + k]) << (8 * k);
484 }
485 } else if (field.datatype == pcl::PCLPointField::UINT8 && field.count == 4) {
486 ti = 0;
487 for (size_t k = 0; k < 4; ++k) {
488 ti |= static_cast<uint32_t>(output_cloud->data[point_offset + k]) << (8 * k);
489 }
490 } else {
491 std::cerr << "Error: Unknown datatype for timestamp field" << std::endl;
492 return *this;
493 }
494 }
495 if (tf > 0.0) {
496 time_float.push_back(tf);
497 } else if (ti > 0) {
498 time_int.push_back(ti);
499 }
500 }
501 }
502
503 if (!time_float.empty()) {
504 saveTimestamps<double>(time_float, file_path);
505
506 // Basic statistics
507 std::cout << "Float timestamps: " << time_float.size() << " entries" << std::endl;
508 if (time_float.size() > 1) {
509 double min_ts = *std::min_element(time_float.begin(), time_float.end());
510 double max_ts = *std::max_element(time_float.begin(), time_float.end());
511 std::cout << " Min: " << min_ts << "\n Max: " << max_ts
512 << "\n Duration: " << (max_ts - min_ts) << std::endl;
513 }
514 }
515
516 if (!time_int.empty()) {
517 saveTimestamps<size_t>(time_int, file_path);
518
519 // Basic statistics
520 std::cout << "Integer timestamps: " << time_int.size() << " entries" << std::endl;
521 if (time_int.size() > 1) {
522 uint64_t min_ts = *std::min_element(time_int.begin(), time_int.end());
523 uint64_t max_ts = *std::max_element(time_int.begin(), time_int.end());
524 std::cout << " Min: " << min_ts << "\n Max: " << max_ts
525 << "\n Duration: " << (max_ts - min_ts) << std::endl;
526 }
527 }
528
529 return *this;
530 }

639 bool load_txt(const std::string & file_path)
640 {
641 std::ifstream file(file_path);
642 if (!file.is_open()) {
643 std::cerr << "Error: Failed to open TXT file: " << file_path << std::endl;
644 return false;
645 }
646
647 std::string line;
648 std::vector<std::string> field_names;
649 std::vector<std::vector<float>> data;
650
651 // Read first line to check for header
652 if (!std::getline(file, line)) {
653 std::cerr << "Error: Empty TXT file: " << file_path << std::endl;
654 return false;
655 }
656
657 // Try to parse first line as header (check if it contains non-numeric values)
658 std::istringstream first_line(line);
659 std::string token;
660 bool is_header = false;
661
662 while (first_line >> token) {
663 // Check if token is a number
664 char* end;
665 std::strtof(token.c_str(), &end);
666 if (*end != '\0') {
667 // Not a number, treat as header
668 is_header = true;
669 }
670 field_names.push_back(token);
671 }
672
673 if (!is_header) {
674 // First line was data, use default field names and parse as data
675 field_names.clear();
676 size_t num_fields = 0;
677
678 // Count fields and parse first line as data
679 std::istringstream data_line(line);
680 std::vector<float> point_data;
681 float value;
682 while (data_line >> value) {
683 point_data.push_back(value);
684 num_fields++;
685 }
686 data.push_back(point_data);
687
688 // Generate default field names based on number of fields
689 if (num_fields >= 3) {
690 field_names = {"x", "y", "z"};
691 for (size_t i = 3; i < num_fields; ++i) {
692 field_names.push_back("field_" + std::to_string(i));
693 }
694 } else {
695 for (size_t i = 0; i < num_fields; ++i) {
696 field_names.push_back("field_" + std::to_string(i));
697 }
698 }
699 }
700
701 // Read remaining lines as data
702 while (std::getline(file, line)) {
703 if (line.empty() || line[0] == '#') continue;
704
705 std::istringstream iss(line);
706 std::vector<float> point_data;
707 float value;
708
709 while (iss >> value) {
710 point_data.push_back(value);
711 }
712
713 if (point_data.size() == field_names.size()) {
714 data.push_back(point_data);
715 }
716 }
717
718 if (data.empty()) {
719 std::cerr << "Error: No valid points in TXT file: " << file_path << std::endl;
720 return false;
721 }
722
723 // Build PCLPointCloud2 manually
724 const size_t num_points = data.size();
725 const size_t num_fields = field_names.size();
726
727 input_cloud->width = num_points;
728 input_cloud->height = 1;
729 input_cloud->is_dense = true;
730 input_cloud->is_bigendian = false;
731
732 // Setup fields (all as FLOAT32)
733 input_cloud->fields.clear();
734 input_cloud->fields.resize(num_fields);
735 uint32_t offset = 0;
736
737 for (size_t i = 0; i < num_fields; ++i) {
738 input_cloud->fields[i].name = field_names[i];
739 input_cloud->fields[i].offset = offset;
740 input_cloud->fields[i].datatype = pcl::PCLPointField::FLOAT32;
741 input_cloud->fields[i].count = 1;
742 offset += sizeof(float);
743 }
744
745 input_cloud->point_step = offset;
746 input_cloud->row_step = input_cloud->point_step * input_cloud->width;
747 input_cloud->data.resize(input_cloud->row_step * input_cloud->height);
